# =============================================================================
# Tips for Choosing Parameters
# =============================================================================
#
# input_bit:
#   - 8: Fast generation, good for testing
#   - 16: Balanced, good for most training
#   - 32: More complex, longer trajectories
#   - 64+: Production scale, may take time
#
# num_trajectories:
#   - 5-10: Quick testing
#   - 50-100: Small to medium training
#   - 200-500: Large-scale training
#   - 1000+: Production datasets
#
# max_steps:
#   - 10-20: Short trajectories, fast
#   - 30-50: Medium length, balanced
#   - 100+: Long optimization paths
#
# sample_strategy:
#   - random: Diverse, good for most cases
#   - best: Optimal paths, expert demonstrations
#   - all_valid: Complete coverage, large datasets
#
# level_bound_delta:
#   - 0-1: Tight constraints, near-optimal
#   - 2-3: Moderate flexibility, balanced
#   - 4+: High flexibility, more exploration
#
# init_type:
#   - sklansky: Near-optimal starting point
#   - brent_kung: More room for optimization
#   - normal: Maximum optimization potential
